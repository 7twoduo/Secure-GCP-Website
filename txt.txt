####         CDN: Content Delivery Network

# 3. Create the Backend Service and Enable CDN
resource "google_compute_backend_service" "default" {
  name                  = "lizzo-monster"
  protocol              = "HTTP"
  port_name             = "http"
  load_balancing_scheme = "EXTERNAL_MANAGED"
  timeout_sec           = 10

  health_checks = [google_compute_health_check.default.id]
  # This enables CDN for the backend service, which is a lot easier than AWS
  enable_cdn = true
  cdn_policy {
    cache_mode  = "CACHE_ALL_STATIC"
    client_ttl  = 3600
    default_ttl = 3600
    max_ttl     = 86400

    cache_key_policy {
      include_host         = true
      include_protocol     = true
      include_query_string = false
    }
  }

  # 4. Associate the WAF Security Policy with the Backend Service
  # This is essentially the WAF For the system, it is a lot different from AWS
  security_policy = google_compute_security_policy.waf_policy.id

  # Attach your existing Instance Group or Network Endpoint Group here
  backend {
    group           = google_compute_instance_group_manager.default.instance_group
    balancing_mode  = "UTILIZATION"
    capacity_scaler = 1.0
  }
}

# Rate Limiting Ban
resource "google_compute_security_policy_rule" "rate_limit_per_ip" {
  security_policy = google_compute_security_policy.waf_policy.name
  priority        = 2000
  description     = "Rate limit requests per source IP"
  action          = "rate_based_ban"

  match {
    versioned_expr = "SRC_IPS_V1"

    config {
      src_ip_ranges = ["*"]
    }
  }

  rate_limit_options {
    conform_action = "allow"
    exceed_action  = "deny(429)"
    enforce_on_key = "IP"

    rate_limit_threshold {
      count        = 120
      interval_sec = 60
    }

    ban_duration_sec = 300

    ban_threshold {
      count        = 500
      interval_sec = 300
    }
  }
}

  match {
    expr {
      expression = <<-EOT
        origin.region_code == 'RU' ||
        origin.region_code == 'CN' ||
        origin.region_code == 'KP' ||
        origin.region_code == 'IR'
      EOT
    }
